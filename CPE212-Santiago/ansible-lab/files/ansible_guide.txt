---
- name: Install common packages
  hosts: all
  become: yes
  gather_facts: yes
  vars:
    common_pkgs:
      - git
      - curl
      - vim
  tasks:
    - name: Update apt cache (Ubuntu)
      ansible.builtin.apt:
        update_cache: yes
        cache_valid_time: 3600
      when: ansible_os_family == "Debian"
      tags: packages

    - name: Install packages (Ubuntu)
      ansible.builtin.apt:
        name: "{{ common_pkgs }}"
        state: present
      when: ansible_os_family == "Debian"
      tags: packages

    - name: Install packages (CentOS)
      ansible.builtin.dnf:          # use yum on older CentOS 7
        name: "{{ common_pkgs }}"
        state: latest
      when: ansible_os_family == "RedHat"
      tags: packages

    - name: Distro-agnostic install
      ansible.builtin.package:
        name: htop
        state: present

    - name: Remove a package
      ansible.builtin.apt:
        name: telnet
        state: absent
        purge: yes
        autoremove: yes
      when: ansible_os_family == "Debian"

    - name: Install .deb from URL
      ansible.builtin.apt:
        deb: https://example.com/pkg.deb
      when: ansible_os_family == "Debian"

    - name: Install pip package
      ansible.builtin.pip:
        name: docker
        state: present

---
- name: File management
  hosts: all
  become: yes
  vars:
    app_port: 8080
    app_name: myapp
  tasks:
    - name: Create directory
      ansible.builtin.file:
        path: /opt/{{ app_name }}
        state: directory
        owner: root
        group: root
        mode: '0755'
        recurse: yes

    - name: Create empty file / touch
      ansible.builtin.file:
        path: /opt/{{ app_name }}/app.log
        state: touch
        mode: '0644'

    - name: Create symlink
      ansible.builtin.file:
        src: /opt/{{ app_name }}
        dest: /opt/current
        state: link

    - name: Delete file
      ansible.builtin.file:
        path: /tmp/old.txt
        state: absent

    - name: Copy file from control node
      ansible.builtin.copy:
        src: files/motd.txt           # looks in ./files or role files/
        dest: /etc/motd
        owner: root
        mode: '0644'
        backup: yes

    - name: Copy inline content
      ansible.builtin.copy:
        content: "Managed by Ansible\n"
        dest: /etc/ansible_managed.txt

    - name: Template (Jinja2) -> templates/app.conf.j2
      ansible.builtin.template:
        src: app.conf.j2
        dest: /etc/{{ app_name }}.conf
        mode: '0644'
        validate: "nginx -t -c %s"      # optional validation
      notify: Restart app

    - name: Ensure line in file
      ansible.builtin.lineinfile:
        path: /etc/ssh/sshd_config
        regexp: '^#?PermitRootLogin'
        line: 'PermitRootLogin no'
        backup: yes
      notify: Restart sshd

    - name: Remove line
      ansible.builtin.lineinfile:
        path: /etc/hosts
        regexp: '^10\.0\.0\.99'
        state: absent

    - name: Insert block of text
      ansible.builtin.blockinfile:
        path: /etc/hosts
        marker: "# {mark} ANSIBLE MANAGED BLOCK"
        block: |
          192.168.56.102 ubuntu1
          192.168.56.103 ubuntu2
          192.168.56.104 centos1

    - name: Regex replace
      ansible.builtin.replace:
        path: /etc/app.conf
        regexp: 'port=\d+'
        replace: 'port={{ app_port }}'

    - name: Fetch file from remote to control node
      ansible.builtin.fetch:
        src: /etc/hostname
        dest: fetched/
        flat: no

    - name: Check if file exists
      ansible.builtin.stat:
        path: /etc/nginx/nginx.conf
      register: nginx_conf

    - name: Use stat result
      ansible.builtin.debug:
        msg: "nginx.conf exists"
      when: nginx_conf.stat.exists

    - name: Download file
      ansible.builtin.get_url:
        url: https://example.com/file.tar.gz
        dest: /tmp/file.tar.gz
        mode: '0644'
        # checksum: sha256:abcd...

    - name: Extract archive
      ansible.builtin.unarchive:
        src: /tmp/file.tar.gz
        dest: /opt/
        remote_src: yes

    - name: Find files older than 7 days
      ansible.builtin.find:
        paths: /var/log
        patterns: "*.log"
        age: 7d
      register: old_logs

  handlers:
    - name: Restart app
      ansible.builtin.service:
        name: myapp
        state: restarted
    - name: Restart sshd
      ansible.builtin.service:
        name: "{{ 'ssh' if ansible_os_family == 'Debian' else 'sshd' }}"
        state: restarted

        name: bob

        name: "daily backup"

    - name: Set hostname

- name: Security and system
  hosts: all
  become: yes
  tasks:
    # UFW (Ubuntu)
    - name: Allow SSH & HTTP via UFW
      community.general.ufw:
        rule: allow
        port: "{{ item }}"
        proto: tcp
      loop: [22, 80, 443]
      when: ansible_os_family == "Debian"

    - name: Enable UFW
      community.general.ufw:
        state: enabled
        policy: deny
      when: ansible_os_family == "Debian"

    # firewalld (CentOS)
    - name: Open http in firewalld
      ansible.posix.firewalld:
        service: http
        permanent: yes
        immediate: yes
        state: enabled
      when: ansible_os_family == "RedHat"

    - name: Open custom port in firewalld
      ansible.posix.firewalld:
        port: 5601/tcp
        permanent: yes
        immediate: yes
        state: enabled
      when: ansible_os_family == "RedHat"

    - name: SELinux permissive
      ansible.posix.selinux:
        policy: targeted
        state: permissive
      when: ansible_os_family == "RedHat"

    - name: Sysctl (needed by Elasticsearch)
      ansible.posix.sysctl:
        name: vm.max_map_count
        value: '262144'
        state: present
        reload: yes

    - name: Mount NFS
      ansible.posix.mount:
        path: /mnt/share
        src: 192.168.56.104:/srv/share
        fstype: nfs
        opts: defaults
        state: mounted

    - name: Create partition-less FS
      community.general.filesystem:
        fstype: ext4
        dev: /dev/sdb

    - name: Wait for port
      ansible.builtin.wait_for:
        port: 9200
        host: 127.0.0.1

# 9. CONDITIONALS, LOOPS, BLOCK/RESCUE/ALWAYS, ERROR HANDLING
################################################################################
---
- name: Control flow
  hosts: all
  become: yes
  tasks:
    - name: When with and/or
      ansible.builtin.debug:
        msg: "Ubuntu 22 or CentOS"
      when: >
        (ansible_distribution == "Ubuntu" and ansible_distribution_major_version == "22")
        or ansible_distribution == "CentOS"

    - name: Loop list
      ansible.builtin.user:
        name: "{{ item }}"
        state: present
      loop: [user1, user2, user3]

    - name: Loop dict list
      ansible.builtin.user:
        name: "{{ item.name }}"
        groups: "{{ item.groups }}"
      loop:
        - { name: dev1, groups: devops }
        - { name: dev2, groups: devops }

    - name: Loop over dict
      ansible.builtin.debug:
        msg: "{{ item.key }} = {{ item.value }}"
      loop: "{{ {'a': 1, 'b': 2} | dict2items }}"

    - name: Loop with index/label
      ansible.builtin.debug:
        msg: "{{ idx }}: {{ item }}"
      loop: [x, y, z]
      loop_control:
        index_var: idx
        label: "{{ item }}"

    - name: Until retry loop
      ansible.builtin.command: curl -s http://localhost:9200
      register: es
      until: es.rc == 0
      retries: 10
      delay: 5
      changed_when: false

    - name: Run only on first host
      ansible.builtin.debug:
        msg: "first only"
      run_once: true

    - name: Ignore errors
      ansible.builtin.command: /bin/false
      ignore_errors: yes

    - name: Custom failure/changed conditions
      ansible.builtin.shell: grep -c error /var/log/syslog
      register: errs
      failed_when: errs.stdout | int > 100
      changed_when: false

    - name: Assert
      ansible.builtin.assert:
        that:
          - ansible_memtotal_mb >= 2048
        fail_msg: "Need at least 2GB RAM"
        success_msg: "RAM OK"

    - name: Fail explicitly
      ansible.builtin.fail:
        msg: "Unsupported OS"
      when: ansible_os_family not in ['Debian', 'RedHat']

    - name: Block / rescue / always
      block:
        - name: Try risky thing
          ansible.builtin.command: /bin/false
      rescue:
        - name: Recover
          ansible.builtin.debug:
            msg: "Task failed, recovering"
      always:
        - name: Always runs
          ansible.builtin.debug:
            msg: "cleanup"

    - name: Block with shared become/when
      block:
        - ansible.builtin.debug: { msg: "RedHat only block" }
      when: ansible_os_family == "RedHat"
      become: yes

    - name: Include tasks dynamically
      ansible.builtin.include_tasks: tasks/extra.yml

    - name: Import tasks statically
      ansible.builtin.import_tasks: tasks/extra.yml

    - name: Pause
      ansible.builtin.pause:
        seconds: 3

    - name: Reboot
      ansible.builtin.reboot:
        reboot_timeout: 300
      when: false      # remove when needed

- name: Rolling update
  hosts: web_servers
  become: yes
  serial: 1                     # one host at a time (or "50%")
  max_fail_percentage: 0
  any_errors_fatal: true
  strategy: linear              # free = hosts proceed independently
  pre_tasks:
    - name: Pre check
      ansible.builtin.debug: { msg: "starting" }
  tasks:
    - name: Delegate to control node
      ansible.builtin.command: echo "{{ inventory_hostname }} updated" >> /tmp/log.txt
      delegate_to: localhost
      become: no

    - name: Long running (async)
      ansible.builtin.command: sleep 30
      async: 60
      poll: 0
      register: job

    - name: Check async job
      ansible.builtin.async_status:
        jid: "{{ job.ansible_job_id }}"
      register: job_result
      until: job_result.finished
      retries: 10
      delay: 5

    - name: Tagged task
      ansible.builtin.debug: { msg: "only with --tags demo" }
      tags: [demo, never_default]

    - name: Always-run tag
      ansible.builtin.debug: { msg: "runs always" }
      tags: always

    - name: Check-mode aware
      ansible.builtin.command: echo real
      check_mode: no            # runs even in --check
  post_tasks:
    - name: Post check
      ansible.builtin.debug: { msg: "done" }

  hosts: db_servers
  become: yes
  vars:
    db_name: appdb
    db_user: appuser
    db_pass: "ChangeMe123!"
  tasks:
    - name: Install MariaDB + python mysql lib
      ansible.builtin.apt:
        name: [mariadb-server, python3-pymysql]
        state: present
        update_cache: yes

    - name: Start MariaDB
      ansible.builtin.service:
        name: mariadb
        state: started
        enabled: yes

    - name: Create database
      community.mysql.mysql_db:
        name: "{{ db_name }}"
        state: present
        login_unix_socket: /var/run/mysqld/mysqld.sock

    - name: Create user
      community.mysql.mysql_user:
        name: "{{ db_user }}"
        password: "{{ db_pass }}"
        priv: "{{ db_name }}.*:ALL"
        host: "%"
        state: present
        login_unix_socket: /var/run/mysqld/mysqld.sock
      no_log: true

    - name: Bind to all interfaces
      ansible.builtin.lineinfile:
        path: /etc/mysql/mariadb.conf.d/50-server.cnf
        regexp: '^bind-address'
        line: 'bind-address = 0.0.0.0'
      notify: Restart mariadb

  handlers:
    - name: Restart mariadb
      ansible.builtin.service:
        name: mariadb
        state: restarted

        update_cache: yes
      when: ansible_os_family == "Debian"

    - name: Install Docker (Ubuntu)
      ansible.builtin.apt:
        name: docker.io
        state: present
      when: ansible_os_family == "Debian"

    - name: Install Docker (CentOS)
      ansible.builtin.dnf:
        name: docker
        state: present
      when: ansible_os_family == "RedHat"

      ansible.builtin.service:
        name: docker
        state: started
        enabled: yes

    - name: Add user to docker group
      ansible.builtin.user:
        name: "{{ ansible_user }}"
        groups: docker
        append: yes

    - name: Install docker python SDK
      ansible.builtin.pip:
        name: docker

    - name: Pull image
      community.docker.docker_image:
        name: "{{ app_image }}"
        source: pull

    - name: Run container
      community.docker.docker_container:
        name: web
        image: "{{ app_image }}"
        state: started
        restart_policy: always
        ports:
          - "8080:80"
        env:
          APP_ENV: production
        volumes:
          - /opt/site:/usr/share/nginx/html:ro
        networks:
          - name: appnet

    - name: Create network
      community.docker.docker_network:
        name: appnet
        state: present

    - name: Create volume
      community.docker.docker_volume:
        name: dbdata

    - name: Run MySQL container with volume
      community.docker.docker_container:
        name: mysql
        image: mysql:8
        state: started
        env:
          MYSQL_ROOT_PASSWORD: "rootpass"
          MYSQL_DATABASE: appdb
        volumes:
          - dbdata:/var/lib/mysql
        networks:
          - name: appnet
      no_log: true

    - name: Build image from Dockerfile
      community.docker.docker_image:
        name: myapp
        tag: v1
        build:
          path: /opt/myapp          # directory containing Dockerfile
        source: build

    - name: Stop container
      community.docker.docker_container:
        name: web
        state: stopped

    - name: Remove container
      community.docker.docker_container:
        name: web
        state: absent

    - name: Docker Compose v2
      community.docker.docker_compose_v2:
        project_src: /opt/myapp
        state: present

    - name: Gather container info
      community.docker.docker_container_info:
        name: web
      register: web_info

    - name: Fallback with plain command
      ansible.builtin.command: docker ps -a
      changed_when: false

# 25. EXAM TIPS / KEY CONCEPTS
################################################################################
# - Ansible is AGENTLESS, push-based, uses SSH (Linux) / WinRM (Windows), YAML playbooks.
# - IDEMPOTENCY: running twice yields same state ("changed=0" 2nd time).
#   command/shell are NOT idempotent unless using creates:/removes:/changed_when.
# - Play = hosts + tasks. Playbook = list of plays. Task = one module call.
# - Handlers run ONCE at end of play, only if notified AND a task reported "changed".
#   Force early: - meta: flush_handlers
# - become: yes = privilege escalation (sudo). become_user: other user.
# - gather_facts: no speeds up plays; facts via ansible_* vars (setup module).
# - FQCN: ansible.builtin.copy (short names also work: copy).
# - YAML: spaces only (no tabs), 2-space indent, "- " for lists, "key: value" for dicts.
#   Quote values with "{{ }}" at start of line:  msg: "{{ var }}".
# - when: takes raw Jinja (no {{ }} braces).  when: var is defined / var | bool.
# - register captures: .stdout .stderr .rc .changed .failed .stdout_lines
# - defaults < vars in role precedence; -e wins over everything.
# - include_* = dynamic (runtime), import_* = static (parse time). Tags/when behave differently.
# - Inventory: static (INI/YAML) or dynamic (plugins/scripts, cloud). Groups of groups via children.
# - Patterns: all, '*', web_servers:db_servers (union), web_servers:&db_servers (intersection),
#   web_servers:!centos1 (exclusion).
# - Debian family: apt, ufw, service "ssh"; RedHat family: dnf/yum, firewalld, service "sshd", group "wheel".
# - Check mode: --check (dry-run), --diff (show file diffs). Some modules (command/shell) skip in check mode.
# - Common failure causes: wrong ansible_user, missing sudo, python not installed
#   (use raw to bootstrap), SSH key not copied (ssh-copy-id), wrong YAML indent,
#   undefined variable, missing collection (ansible.posix / community.general / community.docker).
# - Test connection: ansible all -m ping  -> "pong"; ssh-keygen + ssh-copy-id user@host.
# - Ansible Galaxy: community hub for roles and collections.
# - ansible.cfg search order: ANSIBLE_CONFIG env > ./ansible.cfg > ~/.ansible.cfg > /etc/ansible/ansible.cfg
# - Docker vs VM: containers share host kernel (lightweight); VMs have full guest OS + hypervisor.
# - Image = read-only template; Container = running instance; Volume = persistent data;
#   Registry = image storage (Docker Hub); Dockerfile = build instructions.
