- name: File module patterns
  hosts: web_servers
  tasks:
    - name: Create a directory
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/ansible-directory"
        state: directory                  # file|directory|link|hard|touch|absent

    - name: Create empty file (touch)
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/ansible_test.txt"
        state: touch

    - name: Create symbolic link
      ansible.builtin.file:
        src: "{{ ansible_env.HOME }}/source_file.txt"
        dest: "{{ ansible_env.HOME }}/source_link.txt"
        state: link

    - name: Delete file / directory / symlink
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/source_link.txt"
        state: absent

    - name: Numeric permissions
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/permission_test.txt"
        state: touch
        mode: '0644'                      # ALWAYS quote octal

    - name: Symbolic permissions
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/symbolic-test"
        state: directory
        mode: 'u=rwx,g=rx,o=rx'

    - name: Ownership (needs become)
      become: true
      ansible.builtin.file:
        path: /tmp/ansible-ownership-test
        state: directory
        owner: root
        group: root
        mode: '0755'
        # recurse: yes                    # apply to contents

    - name: Set timestamps
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/timestamp_test.txt"
        state: touch
        access_time: now
        modification_time: "202609080800.00"     # YYYYMMDDHHMM.SS

    - name: Multiple files with loop
      ansible.builtin.file:
        path: "{{ ansible_env.HOME }}/{{ item }}"
        state: touch
      loop: [file1.txt, file2.txt, file3.txt]

    - name: File with content
      ansible.builtin.copy:
        dest: "{{ ansible_env.HOME }}/student_info.txt"
        content: |
          Name: David Owen Santiago
          Course: Computer Engineering

    - name: Copy file from control node (looks in files/ of play or role)
      become: true
      ansible.builtin.copy:
        src: files/default_site.html
        dest: /var/www/html/index.html
        owner: root
        group: root
        mode: '0644'
        # backup: yes | remote_src: yes (copy on the remote itself)

    - name: Template (Jinja2) from templates/
      ansible.builtin.template:
        src: heap.options.j2
        dest: /etc/elasticsearch/jvm.options.d/heap.options
        owner: root
        group: elasticsearch
        mode: "0660"
        # validate: "visudo -cf %s"

    - name: Edit ONE line (regexp + line)
      ansible.builtin.lineinfile:
        path: /etc/ssh/sshd_config
        regexp: '^#?PermitRootLogin'
        line: 'PermitRootLogin no'
        # state: absent removes matching line ; backup: yes

    - name: Insert a block
      ansible.builtin.blockinfile:
        path: /etc/hosts
        marker: "# {mark} ANSIBLE MANAGED"
        block: |
          192.168.56.114 ubuntu1

    - name: Regex replace everywhere in file
      ansible.builtin.replace:
        path: /etc/app.conf
        regexp: 'port=\d+'
        replace: 'port=8080'

    - name: Stat (exists?)
      ansible.builtin.stat:
        path: /etc/nginx/nginx.conf
      register: conf
    - ansible.builtin.debug: { msg: "exists" }
      when: conf.stat.exists

    - name: Download (get_url) / extract (unarchive)
      ansible.builtin.get_url:
        url: https://example.com/file.tar.gz
        dest: /tmp/file.tar.gz
        mode: '0644'
    - ansible.builtin.unarchive:
        src: /tmp/file.tar.gz
        dest: /opt
        remote_src: yes                   # archive already on remote/URL
        creates: /opt/file/bin            # skip if exists (idempotent)
        # extra_opts: [--strip-components=1]

    - name: Read remote file to control node (base64) - used for ES CA cert
      ansible.builtin.slurp:
        src: /etc/elasticsearch/certs/http_ca.crt
      register: es_ca                     # use: {{ es_ca.content | b64decode }}

    - name: fetch (remote -> control node) / find
      ansible.builtin.fetch: { src: /etc/hostname, dest: fetched/ }

# 7. HANDLERS, TAGS, ORDER OF EXECUTION
################################################################################
# - Handlers run ONCE at the END of the play, only if notified AND the task changed.
# - notify: can be a list. Handler name must match exactly.
# - Order: pre_tasks -> (handlers) -> roles -> tasks -> (handlers) -> post_tasks -> (handlers)
# - tags: always (always runs unless --skip-tags always) ; never ; multiple: [a, b]
# - Tags on a role in the play are applied to every task in that role.
---
- name: Handler + tag demo
  hosts: web_servers
  become: true
  tasks:
    - name: Configure
      ansible.builtin.template:
        src: x.j2
        dest: /etc/x.conf
      tags: [config]
      notify:
        - Restart x
        - Reload systemd
  handlers:
    - name: Restart x
      ansible.builtin.systemd: { name: x, state: restarted }
    - name: Reload systemd
      ansible.builtin.systemd: { daemon_reload: true }

    recreate: "{{ image_result is changed }}"

# security.yml (idempotent password reset)
- ansible.builtin.systemd: { name: elasticsearch, daemon_reload: true, enabled: true, state: started }
- ansible.builtin.wait_for: { port: "{{ es_http_port }}", timeout: 180 }
- name: Generate a temporary elastic password (first run only)
  ansible.builtin.command: >
    /usr/share/elasticsearch/bin/elasticsearch-reset-password -u elastic -a -b -s
  args:
    creates: /etc/elasticsearch/.elastic_password_set
  register: es_temp_pw
  retries: 6
  delay: 10
  until: es_temp_pw.rc == 0
  no_log: "{{ es_no_log | default(true) }}"
- name: Set the elastic password to the vaulted value
  ansible.builtin.uri:
    url: "https://localhost:{{ es_http_port }}/_security/user/elastic/_password"
    method: POST
    user: elastic
    password: "{{ es_temp_pw.stdout_lines | last | trim }}"
    force_basic_auth: true
    body_format: json
    body: { password: "{{ es_elastic_password }}" }
    validate_certs: false
    status_code: 200
  when: es_temp_pw is changed
  no_log: true
- ansible.builtin.file: { path: /etc/elasticsearch/.elastic_password_set, state: touch, mode: "0600" }
  when: es_temp_pw is changed
# marker file (creates:) = "run once" trick

# 13. EXAM TIPS + DEBUGGING
################################################################################
# CONCEPTS
# - Agentless, push-based, SSH (Linux) / WinRM (Windows), YAML, needs Python on target (raw bootstraps it).
# - Idempotent: 2nd run shows changed=0. command/shell are NOT idempotent -> add creates:/removes:/changed_when.
# - Play = hosts+tasks ; Playbook = list of plays ; Task = module call ; Role = reusable bundle.
# - Handlers: run once at end, only if a task reported "changed". meta: flush_handlers forces earlier.
# - become: true -> sudo. become_user: other user. become_ask_pass in cfg -> -K equivalent.
# - FQCN (ansible.builtin.copy) preferred; short names (copy, apt) work too.
# - include_* = dynamic (evaluated at runtime) ; import_* = static (parsed up front).
# - Facts: gathered by setup module at play start; use ansible_* ; disable with gather_facts: false.
# - -e (extra vars) beats everything; role defaults are weakest.
# - Dynamic inventory = scripts/plugins (cloud) ; static = INI/YAML.
#
# FAMILY DIFFERENCES (memorize)
#   Debian/Ubuntu: apt, ufw, ssh service, apache2, www-data, /etc/apt, user group "sudo", python3-docker
#   RedHat/CentOS: dnf/yum, firewalld, sshd service, httpd, apache,  /etc/yum.repos.d, group "wheel", SELinux
#   os_family values are "Debian" / "RedHat" (exact case) ; ansible_distribution "Ubuntu" / "CentOS"
#
# YAML GOTCHAS
#   - Spaces only, consistent indent (2). Module args must be INDENTED UNDER the module name.
#   - Quote anything starting with {{ :  msg: "{{ var }}"  ;  quote modes: mode: '0644'
#   - No {{ }} inside when:  (it is already Jinja)
#   - YAML true/false/yes/no are booleans ; "yes" string needs quotes if literal.
#   - A role's tasks/main.yml is a plain list (no "tasks:" header).
#

#   UNREACHABLE            -> wrong ansible_host/ansible_user, SSH not running, ssh-copy-id missing
